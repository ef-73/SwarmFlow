cmake_minimum_required(VERSION 3.8)
project(swarmflow_nav)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
find_package(swarmflow_interfaces REQUIRED)
find_package(tf2 REQUIRED)

# costmap layer: other robots' footprints from /fleet/robot_states (N-05a)
add_library(swarmflow_nav_layers SHARED src/other_robots_layer.cpp src/keep_right_layer.cpp)
target_compile_features(swarmflow_nav_layers PUBLIC cxx_std_17)
target_include_directories(swarmflow_nav_layers PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
ament_target_dependencies(swarmflow_nav_layers rclcpp nav2_costmap_2d pluginlib swarmflow_interfaces tf2)
pluginlib_export_plugin_description_file(nav2_costmap_2d plugins.xml)
install(TARGETS swarmflow_nav_layers LIBRARY DESTINATION lib ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)

install(DIRECTORY launch config behavior_trees DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  file(GLOB _tests RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} test/test_*.py)
  foreach(_t ${_tests})
    get_filename_component(_n ${_t} NAME_WE)
    ament_add_pytest_test(${_n} ${_t} TIMEOUT 120)
  endforeach()
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_prediction_cost test/test_prediction_cost.cpp)
  target_include_directories(test_prediction_cost PRIVATE include)
endif()
ament_package()
