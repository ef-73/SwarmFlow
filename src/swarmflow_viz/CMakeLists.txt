cmake_minimum_required(VERSION 3.8)
project(swarmflow_viz)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)

ament_python_install_package(${PROJECT_NAME})

add_executable(tf_relay_cpp src/tf_relay.cpp)
target_compile_features(tf_relay_cpp PUBLIC cxx_std_17)
ament_target_dependencies(tf_relay_cpp rclcpp tf2_msgs sensor_msgs geometry_msgs)
install(TARGETS tf_relay_cpp DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/viz_node scripts/tf_relay DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  file(GLOB _tests RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} test/test_*.py)
  foreach(_t ${_tests})
    get_filename_component(_n ${_t} NAME_WE)
    ament_add_pytest_test(${_n} ${_t} TIMEOUT 120)
  endforeach()
endif()
ament_package()
